# lavaan 模型（与 AMOS 设定一致）
# 测量模型
PEOU =~ PEOU1 + PEOU2 + PEOU3 + PEOU4
PU =~ PU1 + PU2 + PU3 + PU4
TR =~ TR1 + TR2 + TR3
SAT =~ SAT1 + SAT2 + SAT3
CI =~ CI1 + CI2 + CI3
SAT1 ~~ SAT2

# 结构模型
PEOU =~ PEOU1 + PEOU2 + PEOU3 + PEOU4
PU =~ PU1 + PU2 + PU3 + PU4
TR =~ TR1 + TR2 + TR3
SAT =~ SAT1 + SAT2 + SAT3
CI =~ CI1 + CI2 + CI3
SAT1 ~~ SAT2
PU ~ P1*PEOU
SAT ~ P2*PU + P3*PEOU + P4*TR
CI ~ P5*SAT + P6*PU + P7*TR
H8 := P2*P5
H9 := P1*P2
H10 := P1*P2*P5
TI_PU_CI := P2*P5
TOT_PU_CI := P6 + P2*P5
TI_PEOU_SAT := P1*P2
TOT_PEOU_SAT := P3 + P1*P2
TI_PEOU_CI := P1*P2*P5 + P1*P6 + P3*P5
TOT_PEOU_CI := P1*P2*P5 + P1*P6 + P3*P5
