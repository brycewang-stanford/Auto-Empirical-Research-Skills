# Pinned scientific stack for the CI / Paper-WorkFlow demo gate.
# Three workflows (validate-catalog.yml, sync-aer-skills.yml,
# sync-statspai-skill.yml) used to inline `python3 -m pip install numpy
# pandas matplotlib statsmodels linearmodels` verbatim; pinning the
# versions here lets Dependabot track upstream security/bug-fix releases
# and makes the CI environment reproducible.
#
# Lower bounds chosen to match the versions known to work with the
# 69-Paper-WorkFlow demo gate as of v2026.07. Upper bounds leave room
# for upstream API changes and are not strict pins.
numpy>=2.2.6,<3
pandas>=3.0.5,<4
matplotlib>=3.11.1,<4
statsmodels>=0.14,<1
# Floor raised 4.27 -> 7.0 (2026-08): the 5.x wheels are compiled against the
# NumPy 1.x ABI, so pairing them with the numpy>=2 this file already allows
# makes every import print "A module that was compiled using NumPy 1.x cannot
# be run in NumPy 2.x as it may crash". 7.0 imports clean. It needs Python
# >= 3.10; that is fine here because this file is only installed by the
# validate-catalog and sync-* workflows, which all pin 3.12 — the 3.9 leg of
# the quality matrix installs nothing (every gate it runs is stdlib-only).
linearmodels>=7.0,<8
